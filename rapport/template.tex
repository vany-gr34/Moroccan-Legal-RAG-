\begin{cardinfo}
    Le protocole \textbf{MQTT} (Message Queuing Telemetry Transport) fonctionne sur un modèle 
    \textit{publish/subscribe}. Un broker central reçoit et redistribue les messages entre les clients.
    Il est particulièrement adapté aux réseaux à faible bande passante.
\end{cardinfo}

\subsection{Code Arduino — Capteur DHT11}

\begin{codeblock}{sketch\_dht11.ino}
#include <DHT.h>
#include <PubSubClient.h>
#include <WiFi.h>

// --- Configuration réseau ---
const char* ssid     = "MonReseau_IoT";
const char* password = "MotDePasseSecurise";
const char* broker   = "192.168.1.100";

#define DHTPIN  4
#define DHTTYPE DHT11
DHT dht(DHTPIN, DHTTYPE);

WiFiClient   espClient;
PubSubClient client(espClient);

void setup() {
    Serial.begin(115200);
    dht.begin();

    WiFi.begin(ssid, password);
    while (WiFi.status() != WL_CONNECTED) {
        delay(500);
        Serial.print(".");
    }
    Serial.println("\nWiFi connecté : " + WiFi.localIP().toString());

    client.setServer(broker, 1883);
}

void loop() {
    float humidity    = dht.readHumidity();
    float temperature = dht.readTemperature();

    if (isnan(humidity) || isnan(temperature)) {
        Serial.println("Erreur lecture DHT11 !");
        return;
    }

    // Publication MQTT
    String payload = "{\"temp\":" + String(temperature) +
                     ",\"hum\":"  + String(humidity)    + "}";
    client.publish("iot/capteur/01/data", payload.c_str());

    delay(5000); // Envoi toutes les 5 secondes
}
\end{codeblock}

\subsection{Code Python — Souscription MQTT}

\begin{codeblock}[listing options={language=Python}]{subscriber.py}
import paho.mqtt.client as mqtt
import json

BROKER  = "192.168.1.100"
PORT    = 1883
TOPIC   = "iot/capteur/#"

def on_connect(client, userdata, flags, rc):
    print(f"Connecté au broker (code={rc})")
    client.subscribe(TOPIC)

def on_message(client, userdata, msg):
    data = json.loads(msg.payload.decode())
    print(f"[{msg.topic}] Temp={data['temp']}°C  Hum={data['hum']}%")

client = mqtt.Client()
client.on_connect = on_connect
client.on_message = on_message
client.connect(BROKER, PORT, 60)
client.loop_forever()
\end{codeblock}

\section{Définitions et Termes Clés}
% ------------------------------------------

\begin{definitionbox}{MQTT}
    \textbf{Message Queuing Telemetry Transport} est un protocole de messagerie léger
    basé sur le modèle \textit{publish/subscribe}, conçu pour les environnements
    à ressources limitées et les réseaux à faible bande passante ou haute latence.
\end{definitionbox}

\vspace{0.8em}

\begin{definitionbox}{Broker MQTT}
    Serveur central qui reçoit tous les messages publiés par les clients et les 
    redistribue aux abonnés correspondants. Il gère les sessions, les files de messages
    et les niveaux de QoS.
\end{definitionbox}

\section{Listes et Énumérations}

\subsection{Liste à puces personnalisée}

Composants d'une architecture IoT complète :

\begin{itemize}
    \item \textbf{Couche perception} : capteurs, actionneurs, microcontrôleurs
        \begin{itemize}
            \item Capteurs de température, d'humidité, de luminosité
            \item Actionneurs : relais, moteurs, LEDs
        \end{itemize}
    \item \textbf{Couche réseau} : Wi-Fi, Zigbee, LoRa, Bluetooth
    \item \textbf{Couche middleware} : broker MQTT, API Gateway, file de messages
    \item \textbf{Couche application} : tableaux de bord, alertes, stockage de données
\end{itemize}

\subsection{Liste numérotée — Étapes de configuration}

\begin{enumerate}
    \item Installer \texttt{Mosquitto} sur le serveur : \texttt{sudo apt install mosquitto}
    \item Configurer le fichier \texttt{/etc/mosquitto/mosquitto.conf}
    \item Démarrer le service : \texttt{sudo systemctl start mosquitto}
    \item Tester avec un client : \texttt{mosquitto\_pub -t test -m "Hello IoT"}
    \item Vérifier la réception : \texttt{mosquitto\_sub -t test -v}
\end{enumerate}
